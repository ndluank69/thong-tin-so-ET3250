\medskip
\noindent\textbf{Ghi nhớ:}
\begin{itemize}
    \item \textbf{1 Byte = 8 bit.}
    \item \textbf{Hệ thập phân:} Mỗi bậc $\times 1000$.
    \item \textbf{Hệ nhị phân:} Mỗi bậc $\times 1024$.
    \item \textbf{Đổi từ đơn vị lớn sang nhỏ:} Nhân.
    \item \textbf{Đổi từ đơn vị nhỏ sang lớn:} Chia.
\end{itemize}
